cmake_minimum_required(VERSION 3.16)
project(tinyimx_peer_compare LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG -march=native")
find_package(Threads REQUIRED)
set(MUDUO_SOURCE "" CACHE PATH "Pinned Muduo checkout")
if(MUDUO_SOURCE)
  set(MUDUO_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(MUDUO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(${MUDUO_SOURCE} muduo EXCLUDE_FROM_ALL)
  add_executable(muduo_echo echo_adapter.cpp)
  target_compile_definitions(muduo_echo PRIVATE USE_MUDUO)
  target_include_directories(muduo_echo PRIVATE ${MUDUO_SOURCE})
  target_link_libraries(muduo_echo PRIVATE muduo_net muduo_base Threads::Threads)
endif()
